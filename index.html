비트런 손동작 리모컨카메라로 손 모양을 인식해서 블루투스로 마이크로비트에 명령을 보냅니다

블루투스 연결 안 됨

카메라 · 손 인식

카메라를 켜 주세요손을 화면 가운데에 보여 주면 손가락 관절을 찾아 동작을 판단해요.
손 없음

손 인식 모델은 카메라를 켤 때 불러와요. 인터넷 연결이 필요합니다.

1. 블루투스 연결

목록에서 BBC micro:bit [5글자 이름]을 고르고 페어링을 누르세요.
연결이 끊기거나 이상하면 마이크로비트의 A+B 버튼을 같이 눌러 리셋한 뒤 다시 연결을 누르세요.

2. 현재 명령

정지
보낸 데이터: -

손 동작 표

  • ✋전진손바닥을 활짝 펴기forward
  • ✊후진주먹 쥐기backward
  • 👈좌회전검지로 왼쪽 가리키기left
  • 👉우회전검지로 오른쪽 가리키기right
  • 🚫정지손을 내리거나 다른 모양stop

직접 조종 (손 인식이 안 될 때)

키보드 방향키로도 조종돼요. 스페이스바는 정지입니다.

마이크로비트에 넣을 코드 (MakeCode JavaScript)
  1. MakeCode에서 확장 servo, bluetooth를 추가합니다.
  2. ⚙️ → 프로젝트 설정에서 누구나 블루투스로 연결 가능(No Pairing Required)을 켭니다.
  3. JavaScript 탭에 아래 코드를 붙여 넣고 다운로드합니다.
let data = ""
let prevData = ""

function 비트런_전진 () {
    basic.showArrow(ArrowNames.North)
    servos.P1.run(100)
    servos.P2.run(-100)
}
function 비트런_후진 () {
    basic.showArrow(ArrowNames.South)
    servos.P1.run(-100)
    servos.P2.run(100)
}
function 비트런_좌회전 () {
    basic.showArrow(ArrowNames.West)
    servos.P1.run(-100)
    servos.P2.run(-100)
}
function 비트런_우회전 () {
    basic.showArrow(ArrowNames.East)
    servos.P1.run(100)
    servos.P2.run(100)
}
function 비트런_정지 () {
    servos.P1.stop()
    servos.P2.stop()
    basic.showIcon(IconNames.Square)
}

bluetooth.onUartDataReceived(serial.delimiters(Delimiters.NewLine), function () {
    data = bluetooth.uartReadUntil(serial.delimiters(Delimiters.NewLine))
    if (data == prevData) {
        return
    }
    prevData = data
    if (data == "forward") {
        비트런_전진()
    } else if (data == "backward") {
        비트런_후진()
    } else if (data == "left") {
        비트런_좌회전()
    } else if (data == "right") {
        비트런_우회전()
    } else {
        비트런_정지()
    }
})

// A+B 버튼을 같이 누르면 마이크로비트 리셋 (뒷면 리셋 버튼 대신)
input.onButtonPressed(Button.AB, function () {
    비트런_정지()
    basic.showString("R")
    control.reset()
})

// A 버튼: 블루투스 이름(5글자) 보기
input.onButtonPressed(Button.A, function () {
    basic.showString(control.deviceName())
    basic.showIcon(IconNames.Square)
})

bluetooth.onBluetoothConnected(function () {
    basic.showIcon(IconNames.Yes)
})
bluetooth.onBluetoothDisconnected(function () {
    prevData = ""
    비트런_정지()
    basic.showIcon(IconNames.No)
})

basic.showIcon(IconNames.Heart)
servos.P1.setPulse(1500)
servos.P2.setPulse(1500)
비트런_정지()
bluetooth.startUartService()
